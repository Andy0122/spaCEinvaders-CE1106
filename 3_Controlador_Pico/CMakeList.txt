# Versión mínima requerida de CMake
cmake_minimum_required(VERSION 3.13)

# 1. IMPORTAR EL SDK DE LA PICO (Debe ir antes de project())
include(pico_sdk_import.cmake)

# 2. NOMBRE DE TU PROYECTO
project(Controlador_Pico_SpaceInvaders C CXX ASM)

# 3. INICIALIZAR EL SDK
pico_sdk_init()

# 4. DEFINIR TU EJECUTABLE (Todos los .c que vayas a crear)
add_executable(controlador 
    src/main.c
    src/uart_i2c.c
)

# 5. VINCULAR LIBRERÍAS DE LA PICO 
# Aquí activamos stdlib (pines básicos) y los protocolos que pide el profesor
target_link_libraries(controlador 
    pico_stdlib
    hardware_uart
    hardware_i2c
)

# 6. CONFIGURAR USB / UART (Habilitar salida para printf)
pico_enable_stdio_usb(controlador 1)
pico_enable_stdio_uart(controlador 0)

# 7. ¡LA MAGIA! Generar el archivo .uf2 que se arrastra a la Pico
pico_add_extra_outputs(controlador)
